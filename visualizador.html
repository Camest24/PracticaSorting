Partición de Lomuto — pivote = último elemento del rango

partition() / quickSortRec()

pivote = v[fin]
i = inicio - 1
para j = inicio .. fin-1:
  si v[j] < pivote:
    i++; swap(v[i], v[j])
swap(v[i+1], v[fin])
return i+1  // posición final del pivote
--
quickSortRec(v, inicio, fin):
  si inicio >= fin: return
  p = partition(v, inicio, fin)
  quickSortRec(v, inicio, p-1)
  quickSortRec(v, p+1, fin)

heapify() / heapSort()

heapify(v, n, i):
  mayor = i; izq = 2i+1; der = 2i+2
  si v[izq] > v[mayor]: mayor = izq
  si v[der] > v[mayor]: mayor = der
  si mayor != i: swap(v[i],v[mayor])

heapSort(v):
  para i = n/2-1 .. 0: heapify(v,n,i)
  para i = n-1 .. 1:
    swap(v[0], v[i]); heapify(v,i,0)
